package com.example.momentum

import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.items
import androidx.compose.material.icons.Icons
import androidx.compose.material.icons.filled.Add
import androidx.compose.material.icons.filled.Delete
import androidx.compose.material.icons.filled.Refresh
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.text.style.TextDecoration
import androidx.compose.ui.unit.dp
import kotlinx.coroutines.delay
import java.util.Locale
import java.util.UUID

enum class TimerMode(val label: String) {
    FOCUS("FOCUS SPRINT"),
    BREAK("SHORT BREAK")
}

data class TimerPreset(
    val label: String,
    val focusMinutes: Int,
    val breakMinutes: Int
)

data class BrainNote(
    val id: String = UUID.randomUUID().toString(),
    val text: String,
    val isDone: Boolean = false
)

data class DailyHabit(
    val id: String = UUID.randomUUID().toString(),
    val title: String,
    val isCompletedToday: Boolean = false
)

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent {
            MaterialTheme {
                Surface(
                    modifier = Modifier.fillMaxSize(),
                    color = MaterialTheme.colorScheme.background
                ) {
                    MomentumHomeScreen()
                }
            }
        }
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun MomentumHomeScreen() {
    val presets = remember {
        listOf(
            TimerPreset("25/5", 25, 5),
            TimerPreset("50/10", 50, 10)
        )
    }

    var selectedPreset by remember { mutableStateOf(presets.first()) }
    var mode by remember { mutableStateOf(TimerMode.FOCUS) }
    var remainingSeconds by remember { mutableIntStateOf(selectedPreset.focusMinutes * 60) }
    var isRunning by remember { mutableStateOf(false) }

    var showBrainDumpSheet by remember { mutableStateOf(false) }
    var draftNoteText by remember { mutableStateOf("") }
    var brainNotes by remember { mutableStateOf(listOf<BrainNote>()) }

    var habits by remember {
        mutableStateOf(
            listOf(
                DailyHabit(title = "Drink 6 glasses of water"),
                DailyHabit(title = "10-min away-from-screen break"),
                DailyHabit(title = "Plan tomorrow's top 1 task")
            )
        )
    }

    // Countdown tick effect (migrate to ForegroundService + persisted timestamps in production)
    LaunchedEffect(isRunning, mode, selectedPreset) {
        while (isRunning && remainingSeconds > 0) {
            delay(1000L)
            remainingSeconds -= 1
        }
        if (isRunning && remainingSeconds == 0) {
            isRunning = false
            mode = if (mode == TimerMode.FOCUS) TimerMode.BREAK else TimerMode.FOCUS
            remainingSeconds = if (mode == TimerMode.FOCUS) {
                selectedPreset.focusMinutes * 60
            } else {
                selectedPreset.breakMinutes * 60
            }
        }
    }

    Scaffold(
        topBar = {
            TopAppBar(
                title = {
                    Text(
                        text = "Momentum",
                        fontWeight = FontWeight.SemiBold
                    )
                }
            )
        }
    ) { innerPadding ->
        LazyColumn(
            modifier = Modifier
                .fillMaxSize()
                .padding(innerPadding)
                .padding(horizontal = 20.dp),
            verticalArrangement = Arrangement.spacedBy(20.dp),
            contentPadding = PaddingValues(vertical = 16.dp)
        ) {
            // Focus Timer Card
            item {
                ElevatedCard(
                    modifier = Modifier.fillMaxWidth()
                ) {
                    Column(
                        modifier = Modifier
                            .fillMaxWidth()
                            .padding(24.dp),
                        horizontalAlignment = Alignment.CenterHorizontally,
                        verticalArrangement = Arrangement.spacedBy(16.dp)
                    ) {
                        AssistChip(
                            onClick = {
                                if (!isRunning) {
                                    mode = if (mode == TimerMode.FOCUS) TimerMode.BREAK else TimerMode.FOCUS
                                    remainingSeconds = if (mode == TimerMode.FOCUS) {
                                        selectedPreset.focusMinutes * 60
                                    } else {
                                        selectedPreset.breakMinutes * 60
                                    }
                                }
                            },
                            label = { Text(mode.label) }
                        )

                        Text(
                            text = formatSeconds(remainingSeconds),
                            style = MaterialTheme.typography.displayLarge,
                            fontWeight = FontWeight.Bold
                        )

                        Row(
                            horizontalArrangement = Arrangement.spacedBy(8.dp)
                        ) {
                            presets.forEach { preset ->
                                FilterChip(
                                    selected = selectedPreset == preset,
                                    onClick = {
                                        isRunning = false
                                        selectedPreset = preset
                                        mode = TimerMode.FOCUS
                                        remainingSeconds = preset.focusMinutes * 60
                                    },
                                    label = { Text(preset.label) }
                                )
                            }
                        }

                        Row(
                            horizontalArrangement = Arrangement.spacedBy(12.dp),
                            verticalAlignment = Alignment.CenterVertically
                        ) {
                            Button(
                                onClick = { isRunning = !isRunning },
                                modifier = Modifier.height(48.dp)
                            ) {
                                Text(
                                    text = when {
                                        isRunning -> "Pause"
                                        remainingSeconds < (if (mode == TimerMode.FOCUS) selectedPreset.focusMinutes else selectedPreset.breakMinutes) * 60 -> "Resume"
                                        else -> "Start Focus"
                                    }
                                )
                            }

                            OutlinedButton(
                                onClick = {
                                    isRunning = false
                                    remainingSeconds = if (mode == TimerMode.FOCUS) {
                                        selectedPreset.focusMinutes * 60
                                    } else {
                                        selectedPreset.breakMinutes * 60
                                    }
                                },
                                modifier = Modifier.height(48.dp)
                            ) {
                                Icon(Icons.Default.Refresh, contentDescription = "Reset")
                                Spacer(Modifier.width(6.dp))
                                Text("Reset")
                            }
                        }

                        TextButton(onClick = { showBrainDumpSheet = true }) {
                            Icon(Icons.Default.Add, contentDescription = null)
                            Spacer(Modifier.width(4.dp))
                            Text("Brain Dump (${brainNotes.count { !it.isDone }})")
                        }
                    }
                }
            }

            // Daily Micro-Habits Section
            item {
                val completedCount = habits.count { it.isCompletedToday }
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Text(
                        text = "Today's Micro-Habits",
                        style = MaterialTheme.typography.titleMedium,
                        fontWeight = FontWeight.SemiBold
                    )
                    Text(
                        text = "$completedCount / ${habits.size} done",
                        style = MaterialTheme.typography.labelLarge,
                        color = MaterialTheme.colorScheme.primary
                    )
                }
            }

            items(habits, key = { it.id }) { habit ->
                OutlinedCard(
                    onClick = {
                        habits = habits.map {
                            if (it.id == habit.id) it.copy(isCompletedToday = !it.isCompletedToday) else it
                        }
                    },
                    modifier = Modifier.fillMaxWidth()
                ) {
                    Row(
                        modifier = Modifier
                            .fillMaxWidth()
                            .padding(horizontal = 16.dp, vertical = 12.dp),
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Checkbox(
                            checked = habit.isCompletedToday,
                            onCheckedChange = { checked ->
                                habits = habits.map {
                                    if (it.id == habit.id) it.copy(isCompletedToday = checked) else it
                                }
                            }
                        )
                        Spacer(Modifier.width(8.dp))
                        Text(
                            text = habit.title,
                            style = MaterialTheme.typography.bodyLarge,
                            textDecoration = if (habit.isCompletedToday) TextDecoration.LineThrough else TextDecoration.None
                        )
                    }
                }
            }
        }
    }

    if (showBrainDumpSheet) {
        ModalBottomSheet(onDismissRequest = { showBrainDumpSheet = false }) {
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 20.dp, vertical = 12.dp),
                verticalArrangement = Arrangement.spacedBy(12.dp)
            ) {
                Text(
                    text = "Quick Brain Dump",
                    style = MaterialTheme.typography.titleMedium,
                    fontWeight = FontWeight.SemiBold
                )
                Row(
                    verticalAlignment = Alignment.CenterVertically,
                    horizontalArrangement = Arrangement.spacedBy(8.dp)
                ) {
                    OutlinedTextField(
                        value = draftNoteText,
                        onValueChange = { draftNoteText = it },
                        placeholder = { Text("Park a distracting thought...") },
                        singleLine = true,
                        modifier = Modifier.weight(1f)
                    )
                    Button(
                        onClick = {
                            val trimmed = draftNoteText.trim()
                            if (trimmed.isNotEmpty()) {
                                brainNotes = listOf(BrainNote(text = trimmed)) + brainNotes
                                draftNoteText = ""
                            }
                        }
                    ) {
                        Text("Add")
                    }
                }

                brainNotes.forEach { note ->
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Checkbox(
                            checked = note.isDone,
                            onCheckedChange = { checked ->
                                brainNotes = brainNotes.map {
                                    if (it.id == note.id) it.copy(isDone = checked) else it
                                }
                            }
                        )
                        Text(
                            text = note.text,
                            modifier = Modifier.weight(1f),
                            textDecoration = if (note.isDone) TextDecoration.LineThrough else TextDecoration.None
                        )
                        IconButton(
                            onClick = {
                                brainNotes = brainNotes.filterNot { it.id == note.id }
                            }
                        ) {
                            Icon(Icons.Default.Delete, contentDescription = "Delete note")
                        }
                    }
                }

                Spacer(Modifier.height(24.dp))
            }
        }
    }
}

private fun formatSeconds(totalSeconds: Int): String {
    val minutes = totalSeconds / 60
    val seconds = totalSeconds % 60
    return String.format(Locale.US, "%02d:%02d", minutes, seconds)
}
